Copy this script into your Google Sheet's Extensions > Apps Script editor to enable 100% Live Auto-Syncing & Monthly Tab Generation:

function doGet(e) {
  var sheetId = "1IgggnPVkK46NH3sVZCgh2QZdWB5tE9JS3Q_NNQH577M";
  var ss = SpreadsheetApp.openById(sheetId);
  var ledgerSheet = ss.getSheetByName("Ledger");
  
  if (!ledgerSheet) {
    return ContentService.createTextOutput(JSON.stringify({ error: "Sheet 'Ledger' not found!" })).setMimeType(ContentService.MimeType.JSON);
  }
  
  var data = ledgerSheet.getDataRange().getValues();
  var headersIndex = 3; // Row 4 (0-based)
  var headers = data[headersIndex].map(function(h) { return h.toString().trim(); });
  var rows = [];
  
  for (var i = headersIndex + 1; i < data.length; i++) {
    var rowData = data[i];
    var rowObject = {};
    var hasValue = false;
    
    for (var j = 0; j < headers.length; j++) {
      var headerName = headers[j] || ("Col_" + j);
      var cellVal = rowData[j];
      if (cellVal instanceof Date) {
        cellVal = Utilities.formatDate(cellVal, ss.getSpreadsheetTimeZone(), "M/d/yyyy");
      }
      rowObject[headerName] = cellVal;
      if (cellVal !== "" && cellVal !== null) hasValue = true;
    }
    
    if (hasValue && (rowObject["Order #"] || rowObject["Order # "] || rowObject["Order #\n"])) {
      rows.push(rowObject);
    }
  }
  
  return ContentService.createTextOutput(JSON.stringify({ data: rows })).setMimeType(ContentService.MimeType.JSON);
}

function autoCreateMonthTab() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var mainSheet = ss.getSheetByName("Ledger");
  if (!mainSheet) return;
  
  var months = ["January", "February", "March", "April", "May", "June", "July", "August", "September", "October", "November", "December"];
  var now = new Date();
  var currentMonthName = months[now.getMonth()] + " (" + now.getFullYear() + ")";
  
  var monthSheet = ss.getSheetByName(currentMonthName) || ss.insertSheet(currentMonthName);
  monthSheet.clear();
  
  var data = mainSheet.getDataRange().getValues();
  var headers = data[3];
  var filteredRows = [data[0], data[1], data[2], headers];
  
  var dateColIndex = -1;
  for (var k = 0; k < headers.length; k++) {
    if (headers[k].toString().trim().toLowerCase().includes("order date")) { dateColIndex = k; break; }
  }
  
  for (var i = 4; i < data.length; i++) {
    var d = new Date(data[i][dateColIndex]);
    if (!isNaN(d.getTime()) && d.getMonth() === now.getMonth() && d.getFullYear() === now.getFullYear()) {
      filteredRows.push(data[i]);
    }
  }
  
  if (filteredRows.length > 0) {
    monthSheet.getRange(1, 1, filteredRows.length, filteredRows[0].length).setValues(filteredRows);
  }
}
